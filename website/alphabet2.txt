        // Updated dataset for F, G, H, I, J with requested exact vocabulary items:

                    { name: 'Grape', sentence: 'G is for Grape! G - G - Grape!', type: 'grapes' },
                    { name: 'Guitar', sentence: 'G is for Guitar! G - G - Guitar!', type: 'guitar' }
                ]
            },
            'H': {
                letter: 'Hh',
                color: '#0284c7',
                phonics: '/h/ - /aʊ/ - /s/',
                items: [
                    { name: 'House', sentence: 'H is for House! H - H - House!', type: 'house' },
                    { name: 'Hat', sentence: 'H is for Hat! H - H - Hat!', type: 'hat' },
                    { name: 'Horse', sentence: 'H is for Horse! H - H - Horse!', type: 'horse' }
                ]
            },
            'I': {
                letter: 'Ii',
                color: '#ec4899',
                phonics: '/aɪ/ - /s/ - /k/ - /r/ - /iː/ - /m/',
                items: [
                    { name: 'Ice cream', sentence: 'I is for Ice Cream! I - I - Ice Cream!', type: 'icecream' },
                    { name: 'Igloo', sentence: 'I is for Igloo! I - I - Igloo!', type: 'igloo' },
                    { name: 'Island', sentence: 'I is for Island! I - I - Island!', type: 'island' }
                ]
            },
            'J': {
                letter: 'Jj',
                color: '#8b5cf6',
                phonics: '/dʒ/ - /uː/ - /s/',
                items: [
                    { name: 'Juice', sentence: 'J is for Juice! J - J - Juice!', type: 'juice' },
                    { name: 'Jellyfish', sentence: 'J is for Jellyfish! J - J - Jellyfish!', type: 'jellyfish' },
                    { name: 'Jug', sentence: 'J is for Jug! J - J - Jug!', type: 'jug' }
                ]
            }
        };

        let currentLetterKey = 'F';
        let currentItemIndex = 0;
        let isAutoPlaying = false;
        let isPaused = false;
        let autoPlayTimer = null;
        let quizScore = 0;
        let currentQuizQuestion = null;

        // Speech Synthesis configuration
        const synth = window.speechSynthesis;
        let englishVoice = null;

        function loadVoices() {
            if (!synth) return;
            const voices = synth.getVoices();
            englishVoice = voices.find(v => v.lang.startsWith('en') && (v.name.includes('Google') || v.name.includes('Natural') || v.name.includes('Samantha') || v.name.includes('Karen'))) 
                        || voices.find(v => v.lang.startsWith('en')) || null;
        }
        if (speechSynthesis.onvoiceschanged !== undefined) {
            speechSynthesis.onvoiceschanged = loadVoices;
        }

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            try {
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'correct') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(659.25, audioCtx.currentTime + 0.1);
                    osc.frequency.exponentialRampToValueAtTime(783.99, audioCtx.currentTime + 0.2);
                    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.5);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.5);
                } else if (type === 'wrong') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(140, audioCtx.currentTime + 0.3);
                    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.3);
                } else if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(400, audioCtx.currentTime);
                    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.08);
                }
            } catch (e) {
                console.log("Audio play error:", e);
            }
        }

        function speakText(text, callback) {
            if (!synth) return;
            synth.cancel();

            const utterance = new SpeechSynthesisUtterance(text);
            utterance.rate = 0.85;
            utterance.pitch = 1.2;
            if (englishVoice) {
                utterance.voice = englishVoice;
            }

            if (callback) {
                utterance.onend = callback;
            }

            synth.speak(utterance);
        }

        function switchMode(mode) {
            playSound('click');
            if (isAutoPlaying) stopAutoPlay();

            const learnSec = document.getElementById('section-learn');
            const quizSec = document.getElementById('section-quiz');

            if (mode === 'learn') {
                learnSec.classList.remove('hidden');
                learnSec.classList.add('flex');
                quizSec.classList.add('hidden');
                quizSec.classList.remove('flex');
            } else if (mode === 'quiz') {
                quizSec.classList.remove('hidden');
                quizSec.classList.add('flex');
                learnSec.classList.add('hidden');
                learnSec.classList.remove('flex');
                generateQuizQuestion();
            }
        }

        function selectLetter(letterKey) {
            playSound('click');
            currentLetterKey = letterKey;
            currentItemIndex = 0;
            updateLetterDisplay();
        }

        function selectItem(index) {
            playSound('click');
            currentItemIndex = index;
            updateItemDisplay();
            speakCurrentWord();
        }

        function updateLetterDisplay() {
            const data = ALPHABET_DATA[currentLetterKey];

            ['F', 'G', 'H', 'I', 'J'].forEach(key => {
                const btn = document.getElementById(`nav-btn-${key}`);
                if (key === currentLetterKey) {
                    btn.classList.add('ring-4', 'ring-offset-2', 'ring-emerald-400', 'scale-110');
                } else {
                    btn.classList.remove('ring-4', 'ring-offset-2', 'ring-emerald-400', 'scale-110');
                }
            });

            const displayLetter = document.getElementById('display-letter');
            displayLetter.innerText = data.letter;
            displayLetter.style.color = data.color;
            document.getElementById('display-phonics').innerText = data.phonics;

            const gridContainer = document.getElementById('word-selector-grid');
            gridContainer.innerHTML = '';

            data.items.forEach((item, idx) => {
                const card = document.createElement('button');
                card.onclick = () => selectItem(idx);
                const isActive = idx === currentItemIndex;
                
                card.className = `card-pop p-3 rounded-2xl border-2 flex flex-col items-center justify-center gap-1 transition ${
                    isActive ? 'bg-emerald-400 text-white border-emerald-600 shadow-lg scale-105' : 'bg-white text-slate-700 border-slate-200 hover:border-emerald-300'
                }`;
                card.innerHTML = `
                    <span class="text-lg font-black">${item.name}</span>
                    <span class="text-xs ${isActive ? 'text-emerald-100' : 'text-slate-400'}">Click to view</span>
                `;
                gridContainer.appendChild(card);
            });

            updateItemDisplay();
        }

        function updateItemDisplay() {
            const data = ALPHABET_DATA[currentLetterKey];
            const item = data.items[currentItemIndex];

            document.getElementById('active-word-text').innerText = item.name;
            document.getElementById('active-sentence-text').innerText = `"${item.sentence}"`;

            drawCanvasIllustration(item.type);

            const gridContainer = document.getElementById('word-selector-grid');
            Array.from(gridContainer.children).forEach((child, idx) => {
                if (idx === currentItemIndex) {
                    child.className = "card-pop p-3 rounded-2xl border-2 flex flex-col items-center justify-center gap-1 transition bg-emerald-400 text-white border-emerald-600 shadow-lg scale-105";
                } else {
                    child.className = "card-pop p-3 rounded-2xl border-2 flex flex-col items-center justify-center gap-1 transition bg-white text-slate-700 border-slate-200 hover:border-emerald-300";
                }
            });
        }

        function speakCurrentLetter() {
            const data = ALPHABET_DATA[currentLetterKey];
            speakText(`Letter ${currentLetterKey}. Sound: ${data.phonics.replace(/\//g, '')}`);
        }

        function speakCurrentWord() {
            const item = ALPHABET_DATA[currentLetterKey].items[currentItemIndex];
            speakText(item.sentence);
        }

        function drawCanvasIllustration(type) {
            const canvas = document.getElementById('illustrationCanvas');
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const cx = canvas.width / 2;
            const cy = canvas.height / 2;

            ctx.save();

            switch (type) {
                case 'fish':
                    // Orange swimming fish
                    ctx.fillStyle = '#f97316';
                    ctx.beginPath(); ctx.ellipse(cx, cy, 40, 26, 0, 0, Math.PI * 2); ctx.fill();
                    ctx.beginPath(); ctx.moveTo(cx - 30, cy); ctx.lineTo(cx - 58, cy - 22); ctx.lineTo(cx - 58, cy + 22); ctx.close(); ctx.fill();
                    ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 4;
                    ctx.beginPath(); ctx.arc(cx - 5, cy, 22, -Math.PI/3, Math.PI/3); ctx.stroke();
                    drawCuteFace(ctx, cx + 18, cy - 2);
                    break;

                case 'fox':
                    // Cute Orange Fox
                    ctx.fillStyle = '#ea580c';

        function startAutoPlayLesson() {
            playSound('click');
            switchMode('learn');
            isAutoPlaying = true;
            isPaused = false;
            
            document.getElementById('video-control-bar').classList.remove('hidden');
            document.getElementById('video-control-bar').classList.add('flex');

            const btnAuto = document.getElementById('btn-mode-autoplay');
            btnAuto.classList.add('ring-4', 'ring-amber-300');
            document.getElementById('autoplay-btn-label').innerText = "Stop Lesson";
            btnAuto.onclick = stopAutoPlay;

            let letterSequence = ['F', 'G', 'H', 'I', 'J'];
            let currentSeqIdx = 0;

            window.playNextVideoStep = function() {
                if (!isAutoPlaying) return;

                if (currentSeqIdx >= letterSequence.length) {
                    stopAutoPlay();
                    speakText("Great job! You finished learning letters F to J with Prabhath Sir!");
                    return;
                }

                currentLetterKey = letterSequence[currentSeqIdx];
                currentItemIndex = 0;
                updateLetterDisplay();

                document.getElementById('video-progress-text').innerText = `Letter ${currentLetterKey} (${currentSeqIdx + 1} of 5)`;

                speakText(`Letter ${currentLetterKey}`, () => {
                    if (!isAutoPlaying || isPaused) return;
                    speakCurrentWord();
                    
                    autoPlayTimer = setTimeout(() => {
                        if (isPaused) return;
                        currentSeqIdx++;
                        window.playNextVideoStep();
                    }, 5500);
                });
            };

            window.videoSeqIndex = () => currentSeqIdx;
            window.setVideoSeqIndex = (i) => { currentSeqIdx = i; };

            window.playNextVideoStep();
        }

        function togglePauseVideo() {
            isPaused = !isPaused;
            const btn = document.getElementById('btn-pause-video');
            if (isPaused) {
                if (synth) synth.pause();
                if (autoPlayTimer) clearTimeout(autoPlayTimer);
                btn.innerHTML = `<i class="fa-solid fa-play"></i> <span>Resume</span>`;
            } else {
                if (synth) synth.resume();
                btn.innerHTML = `<i class="fa-solid fa-pause"></i> <span>Pause</span>`;
                speakCurrentWord();
            }
        }

        function prevVideoStep() {
            if (!isAutoPlaying) return;
            let idx = window.videoSeqIndex();
            if (idx > 0) {
                if (autoPlayTimer) clearTimeout(autoPlayTimer);
                window.setVideoSeqIndex(idx - 1);
                window.playNextVideoStep();
            }
        }

        function nextVideoStep() {
            if (!isAutoPlaying) return;
            let idx = window.videoSeqIndex();
            if (idx < 4) {
                if (autoPlayTimer) clearTimeout(autoPlayTimer);
                window.setVideoSeqIndex(idx + 1);
                window.playNextVideoStep();
            }
        }

        function stopAutoPlay() {
            isAutoPlaying = false;
            isPaused = false;
            if (autoPlayTimer) clearTimeout(autoPlayTimer);
            if (synth) synth.cancel();

            document.getElementById('video-control-bar').classList.add('hidden');
            document.getElementById('video-control-bar').classList.remove('flex');

            const btnAuto = document.getElementById('btn-mode-autoplay');
            btnAuto.classList.remove('ring-4', 'ring-amber-300');
            document.getElementById('autoplay-btn-label').innerText = "Auto Play Lesson";
            btnAuto.onclick = startAutoPlayLesson;
        }

        function generateQuizQuestion() {
            const keys = ['F', 'G', 'H', 'I', 'J'];
            const targetKey = keys[Math.floor(Math.random() * keys.length)];
            const data = ALPHABET_DATA[targetKey];
            const targetItem = data.items[Math.floor(Math.random() * data.items.length)];

            const otherItems = [];
            keys.filter(k => k !== targetKey).forEach(k => {
                ALPHABET_DATA[k].items.forEach(it => otherItems.push({ ...it, letterKey: k }));
            });

            otherItems.sort(() => 0.5 - Math.random());
            const options = [
                { ...targetItem, letterKey: targetKey, isCorrect: true },
                { ...otherItems[0], isCorrect: false },
                { ...otherItems[1], isCorrect: false }
            ].sort(() => 0.5 - Math.random());

            currentQuizQuestion = {
                letter: targetKey,
                targetItem: targetItem,
                options: options
            };

            document.getElementById('quiz-question-title').innerText = `Find the picture for Letter ${targetKey}!`;
            document.getElementById('quiz-feedback').innerText = '';

            const optionsContainer = document.getElementById('quiz-options-container');
            optionsContainer.innerHTML = '';

            options.forEach((opt, idx) => {
                const btn = document.createElement('button');
                btn.className = "card-pop bg-white hover:bg-rose-50 p-4 rounded-2xl border-4 border-rose-200 shadow-lg flex flex-col items-center justify-center gap-3 transition transform active:scale-95";
                
                const canvasId = `quiz-canvas-${idx}`;
                btn.innerHTML = `
                    <div class="w-24 h-24 bg-rose-50 rounded-xl flex items-center justify-center overflow-hidden">
                        <canvas id="${canvasId}" width="90" height="90"></canvas>
                    </div>
                    <span class="text-xl font-black text-slate-800">${opt.name}</span>
                `;

                btn.onclick = () => handleQuizAnswer(opt.isCorrect, opt.name);
                optionsContainer.appendChild(btn);

                setTimeout(() => {
                    const c = document.getElementById(canvasId);
                    if (c) {
                        const ctx = c.getContext('2d');
                        ctx.scale(0.53, 0.53);
                        drawCanvasIllustrationOnCtx(ctx, opt.type);
                    }
                }, 50);
            });

            playQuizPromptAudio();
        }

        function drawCanvasIllustrationOnCtx(ctx, type) {

            ctx.translate(-5, -5);
            ctx.restore();
        }

        function playQuizPromptAudio() {

        }

        function handleQuizAnswer(isCorrect, name) {
            const feedbackEl = document.getElementById('quiz-feedback');
            if (isCorrect) {
                playSound('correct');
                quizScore += 10;
                document.getElementById('quiz-score').innerText = `Score: ${quizScore}`;
                feedbackEl.className = "mt-6 text-2xl font-black text-emerald-600 animate-bounce";
                feedbackEl.innerText = `🎉 Wonderful! ${name} starts with ${currentQuizQuestion.letter}!`;
                
                triggerConfetti();
                speakText(`Wonderful! ${name} starts with ${currentQuizQuestion.letter}!`);

                setTimeout(() => {
                    generateQuizQuestion();
                }, 2500);
            } else {
                playSound('wrong');
                feedbackEl.className = "mt-6 text-xl font-bold text-rose-500 animate-shake";
                feedbackEl.innerText = `Oops! Try again! Find the item for ${currentQuizQuestion.letter}.`;
                speakText(`Oops! Try again!`);
            }
        }

        function resetQuiz() {

        function triggerConfetti() {
            const canvas = document.getElementById('confettiCanvas');
            const ctx = canvas.getContext('2d');
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;

            const particles = [];
            const colors = ['#f43f5e', '#3b82f6', '#10b981', '#f59e0b', '#8b5cf6'];

            for (let i = 0; i < 70; i++) {
                particles.push({
                    x: canvas.width / 2,
                    y: canvas.height / 2,
                    vx: (Math.random() - 0.5) * 14,
                    vy: (Math.random() - 0.7) * 14,
                    size: Math.random() * 10 + 6,
                    color: colors[Math.floor(Math.random() * colors.length)],
                    rotation: Math.random() * 360,
                    life: 100
                });
            }

            function animateConfetti() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                let alive = false;

                particles.forEach(p => {
                    if (p.life > 0) {
                        alive = true;
                        p.x += p.vx;
                        p.y += p.vy;
                        p.vy += 0.3;
                        p.life -= 1.5;

                        ctx.save();
                        ctx.translate(p.x, p.y);
                        ctx.rotate((p.rotation * Math.PI) / 180);
                        ctx.fillStyle = p.color;
                        ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
                        ctx.restore();
                    }
                });

                if (alive) {
                    requestAnimationFrame(animateConfetti);
                } else {
                    ctx.clearRect(0, 0, canvas.width, canvas.height);
                }
            }

            animateConfetti();
        }
